cmake_minimum_required(VERSION 3.25)
project(ahq_general_audio VERSION 0.1.0 LANGUAGES CXX)

add_library(ahq_audio_core STATIC src/gainProcessor.cpp)
add_library(ahq::audio_core ALIAS ahq_audio_core)
target_include_directories(ahq_audio_core PUBLIC include)
target_compile_features(ahq_audio_core PUBLIC cxx_std_20)
set_target_properties(ahq_audio_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)

include(CTest)
if(BUILD_TESTING)
    add_executable(gainProcessor_test tests/gainProcessor_test.cpp)
    target_link_libraries(gainProcessor_test PRIVATE ahq_audio_core)
    set_target_properties(gainProcessor_test PROPERTIES CXX_EXTENSIONS OFF)
    add_test(NAME gainProcessor_test COMMAND gainProcessor_test)
endif()

if(MSVC)
    target_compile_options(ahq_audio_core PRIVATE /W4)
    if(BUILD_TESTING)
        target_compile_options(gainProcessor_test PRIVATE /W4)
    endif()
else()
    target_compile_options(ahq_audio_core PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_TESTING)
        target_compile_options(gainProcessor_test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
