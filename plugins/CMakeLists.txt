# Header-only shared native UI; linked into each plugin, with no extra runtime DLL.
add_library(ahq_plugin_ui INTERFACE)
add_library(ahq::plugin_ui ALIAS ahq_plugin_ui)
target_compile_features(ahq_plugin_ui INTERFACE cxx_std_20)
target_sources(ahq_plugin_ui INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/common/ui/pluginTheme.h
    ${CMAKE_CURRENT_SOURCE_DIR}/common/ui/presetLibrary.h
    ${CMAKE_CURRENT_SOURCE_DIR}/common/ui/presetBar.h)

get_property(ahq_plugin_directories DIRECTORY PROPERTY SUBDIRECTORIES)
foreach(plugin_directory IN LISTS ahq_plugin_directories)
    get_property(plugin_targets DIRECTORY ${plugin_directory} PROPERTY BUILDSYSTEM_TARGETS)
    foreach(plugin_target IN LISTS plugin_targets)
        get_target_property(plugin_type ${plugin_target} TYPE)
        if(plugin_type STREQUAL "STATIC_LIBRARY")
            target_link_libraries(${plugin_target} PRIVATE ahq::plugin_ui)
        endif()
    endforeach()
endforeach()
